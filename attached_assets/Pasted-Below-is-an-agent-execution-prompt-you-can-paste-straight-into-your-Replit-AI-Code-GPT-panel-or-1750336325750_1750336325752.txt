Edit `src/layer3_frameworks/flask_app/__init__.py`:

```python
from flask import Flask, send_from_directory
from gradio_app.vision_tools import demo as gradio_ui

def create_app():
  app = Flask(__name__, static_folder="../vue_frontend/dist/assets")

  # ---- Blueprints ----
  from layer2_interfaces.controllers import agent_bp, identity_bp, mission_bp, roadmap_bp
  for bp in (agent_bp, identity_bp, mission_bp, roadmap_bp):
      app.register_blueprint(bp)

  # ---- Gradio UI mounted on /tools/vision ----
  try:
      gradio_ui.mount_to_app(app, path="/tools/vision")
  except Exception as exc:
      app.logger.warning("Gradio not available %s", exc)

  # ---- Vue single-page front-end -------------
  @app.route("/", defaults={"path": ""})
  @app.route("/<path:path>")
  def vue_root(path):
      # return the built index.html from vite
      return send_from_directory("../vue_frontend/dist", "index.html")

  return app

3.  GRADIO MINI-APP  (no NumPy source import)

In gradio_app/vision_tools.py

import gradio as gr
from layer2_interfaces.gateways.cv_gateway import CVGateway

cv = CVGateway()

with gr.Blocks(title="Missiora Vision Tools") as demo:
    with gr.Tab("Caption"):
        img = gr.Image()
        out = gr.Textbox()
        btn = gr.Button("Describe")
        btn.click(lambda i: cv.caption(i), img, out)
    with gr.Tab("Detect"):
        img2 = gr.Image()
        out2 = gr.JSON()
        btn2 = gr.Button("Detect")
        btn2.click(lambda i: cv.detect(i), img2, out2)

def mount_to_app(app, path="/tools/vision"):
    app = gr.mount_gradio_app(app, demo, path)
    return app
