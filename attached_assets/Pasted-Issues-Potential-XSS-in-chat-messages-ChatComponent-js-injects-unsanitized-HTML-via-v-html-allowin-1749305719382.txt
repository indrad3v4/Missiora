# 6. Ensure components served
grep -q "vue-build.js" static/js/vue-app.js || cat >> static/js/vue-app.js <<'EOF'

import IdentityView from '/src/ui/components/IdentityView.vue'
import MissionView from '/src/ui/components/MissionView.vue'
import RoadmapView from '/src/ui/components/RoadmapView.vue'

createApp({
  components:{IdentityView,MissionView,RoadmapView},
  data(){return{tab:'Identity', milestones:[]}},
  mounted(){
    fetch('/roadmap').then(r=>r.json()).then(data=>{this.milestones=data})
  },
  template:`
    <header>{{ tab }}</header>
    <IdentityView v-if="tab==='Identity'" @save="saveIdentity"/>
    <MissionView v-if="tab==='Mission'" @save="saveMission"/>
    <RoadmapView v-if="tab==='Roadmap'" :milestones="milestones" @complete="complete"/>
    <footer>
      <button @click="tab='Identity'">Identity</button>
      <button @click="tab='Mission'">Mission</button>
      <button @click="tab='Roadmap'">Roadmap</button>
    </footer>
  `,
  methods:{
    saveIdentity(data){fetch('/identity',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({identity:data})})},
    saveMission(text){fetch('/mission',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({mission:text})})},
    complete(id){fetch('/milestone/'+id+'/complete',{method:'POST'}).then(()=>location.reload())}
  }
}).mount('#app')
EOF
