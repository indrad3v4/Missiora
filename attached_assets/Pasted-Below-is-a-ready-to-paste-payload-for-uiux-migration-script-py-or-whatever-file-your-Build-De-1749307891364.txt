Below is a **ready-to-paste* payload for `uiux_migration_script.py`* (or whatever file your Build/Dev-Agent will execute).

It performs **one clean migration pass**:

* **Re-writes the header, footer, and three core views as proper Vue SFCs** inside `static/src/ui/components/`.
* **Injects orchestrator helpers** that:

  * pre-fill Identity fields from `/api/profile`,
  * let the Mission view call `/api/orchestrator/reflect` for live guidance,
  * let the Roadmap view pull / post milestones.
* **Upgrades the Flask routes** (adds `/api/profile`, `/api/mission`, `/api/roadmap`) and the SQLAlchemy model `Milestone`.
* **Seeds a default roadmap** derived from Identity + Mission.
* Runs a lightweight Vue build (`npm run build`) and copies `dist/` into `static/`.
* Leaves you with a crisp, gradient header, a neumorphic footer, and the **lifeline Roadmap component** that scroll-locks the active milestone (matching your fourth mock).

---

```python
"""
uiux_migration_script.py  — single-shot migration to “Identity | Mission | Roadmap”
Run:  python uiux_migration_script.py
"""

import os, shutil, subprocess, textwrap, pathlib, time
from pathlib import Path

# ────────────────────────────────────────────
# 1.  Scaffolding – folders
# ────────────────────────────────────────────
ROOT = Path(__file__).parent
VUE_PATH = ROOT / "static" / "src" / "ui" / "components"
for p in [
    VUE_PATH,
    ROOT / "static" / "src" / "assets",
    ROOT / "migrations",
]:
    p.mkdir(parents=True, exist_ok=True)

# ────────────────────────────────────────────
# 2.  New SQLAlchemy model & migration stub
# ────────────────────────────────────────────
model_patch = """
# models.py  (append at end)

class Milestone(db.Model):
    __tablename__ = "milestones"
    id          = db.Column(db.Integer, primary_key=True)
    user_id     = db.Column(db.Integer, db.ForeignKey("users.id"), nullable=False)
    title       = db.Column(db.String(128))
    description = db.Column(db.Text)
    position    = db.Column(db.Integer)      # order in lifeline
    status      = db.Column(db.String(16))   # pending | active | done
    created_at  = db.Column(db.DateTime, default=datetime.utcnow)
"""
with open(ROOT / "models.py", "a", encoding="utf-8") as fh:
    fh.write(model_patch)

(Path("migrations/README.md")).write_text(
    "# Manual step: run `flask db migrate && flask db upgrade` after this script\n",
    "utf-8",
)

# ────────────────────────────────────────────
# 3.  Flask API endpoints (very small)
# ────────────────────────────────────────────
routes_patch = """
# routes.py  (append below existing routes)

@bp.route("/api/profile", methods=["GET", "POST"])
@login_required
def profile_api():
    user = current_user
    if request.method == "POST":
        data = request.json
        user.archetype   = data.get("archetype", user.archetype)
        user.location    = data.get("location",  user.location)
        user.current_mood= data.get("mood",      user.current_mood)
        db.session.commit()
        return jsonify({"ok": True})
    return jsonify(user.serialize())  

@bp.route("/api/mission", methods=["GET", "POST"])
@login_required
def mission_api():
    user = current_user
    if request.method == "POST":
        data = request.json
        user.mission_statement = data["mission"]
        user.highest_goal      = data["goal"]
        user.blocks            = data["blocks"]
        db.session.commit()
        from orchestrator import build_default_roadmap
        build_default_roadmap(user)
        return jsonify({"ok": True})
    return jsonify(
        mission=user.mission_statement,
        goal=user.highest_goal,
        blocks=user.blocks,
    )

@bp.route("/api/roadmap", methods=["GET", "PATCH"])
@login_required
def roadmap_api():
    if request.method == "PATCH":
        m_id = request.json["milestone_id"]
        new  = request.json["status"]
        ms   = Milestone.query.get(m_id)
        ms.status = new
        db.session.commit()
    miles = [
        m.as_dict() for m in Milestone.query.filter_by(user_id=current_user.id).order_by(Milestone.position)
    ]
    return jsonify(miles)
"""
with open(ROOT / "routes.py", "a", encoding="utf-8") as fh:
    fh.write(routes_patch)

# ────────────────────────────────────────────
# 4.  Orchestrator helper to build roadmap
# ────────────────────────────────────────────
orch_patch = """
# orchestrator.py  (append)

def build_default_roadmap(user):
    if Milestone.query.filter_by(user_id=user.id).count():
        return  # already exists
    base = [
        ("Define Identity", "Shape archetype, values, expression"),
        ("Build Mission",   "Articulate highest goal + mission"),
        ("Design First Offer", "Craft an offer aligned with mission"),
        ("Secure First Clients", "Close first 3 paid clients"),
        ("Earn €1K",        "Achieve first €1k revenue milestone"),
    ]
    for idx, (t, d) in enumerate(base, start=1):
        db.session.add(
            Milestone(user_id=user.id, title=t, description=d, position=idx,
                      status="active" if idx == 1 else "pending")
        )
    db.session.commit()
"""
with open(ROOT / "orchestrator.py", "a", encoding="utf-8") as fh:
    fh.write(orch_patch)

# ────────────────────────────────────────────
# 5.  Vue SFC templates
# ────────────────────────────────────────────
IDENTITY_VUE = VUE_PATH / "IdentityView.vue"
MISSION_VUE  = VUE_PATH / "MissionView.vue"
ROADMAP_VUE  = VUE_PATH / "RoadmapView.vue"
HEADER_VUE   = VUE_PATH / "BaseHeader.vue"
FOOT_VUE     = VUE_PATH / "BaseFooter.vue"

HEADER_VUE.write_text(
    textwrap.dedent("""\
    <template>
      <header class='gradient-bar flex justify-between items-center p-4 text-white'>
        <h1 class='font-bold tracking-wide'>MISSIORA</h1>
        <div v-if='profile' class='bg-white/20 backdrop-blur rounded-xl px-3 py-1 text-sm'>
          {{ profile.name }} • 🎯 {{ profile.goal or 'Set Your Goal' }} • 📍 {{ profile.location or 'Location' }} • 🕒 {{ profile.quarter or 'Q?' }} • 💰 €{{ profile.cash or 0 }}
        </div>
      </header>
    </template>
    <script setup>
    import { onMounted, ref } from 'vue'
    const profile = ref(null)
    onMounted(async()=>{
      profile.value = await fetch('/api/profile').then(r=>r.json())
    })
    </script>
    <style scoped>
    .gradient-bar{background:linear-gradient(135deg,#4ade80,#3b82f6);}
    </style>
    """),
    "utf-8",
)

FOOT_VUE.write_text(
    textwrap.dedent("""\
    <template>
      <nav class='fixed bottom-0 inset-x-0 flex bg-white border-t'>
        <button v-for='tab in tabs'
                :key='tab'
                @click='$emit(\"change\",tab)'
                :class='[\"flex-1 py-3\", active===tab?\"bg-emerald-400 text-white\":\"\"]'>
          {{ tab }}
        </button>
      </nav>
    </template>
    <script setup>
    const props = defineProps({active:String})
    const tabs = ['Identity','Mission','Roadmap']
    </script>"""),
    "utf-8",
)

IDENTITY_VUE.write_text(
    textwrap.dedent("""\
    <template>
      <div class='p-6 space-y-4'>
        <h2 class='text-xl font-semibold'>Define Your Identity</h2>
        <!-- each input two-way bound to identity object -->
        <input v-model='identity.name' placeholder='Your Name' class='input'/>
        <select v-model='identity.archetype' class='input'>
          <option>Builder</option><option>Guide</option><option>Rebel</option>
        </select>
        <textarea v-model='identity.mood' placeholder='Current Mood' class='input'></textarea>
        <input v-model='identity.location' placeholder='Region/Location' class='input'/>
        <button @click='save' class='btn w-full'>Save Identity</button>
      </div>
    </template>
    <script setup>
    import { ref, onMounted } from 'vue'
    const identity = ref({}); onMounted(async()=>{identity.value =
      await fetch('/api/profile').then(r=>r.json())})
    const save = ()=>fetch('/api/profile',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(identity.value)});
    </script>
    <style scoped>.input{width:100%;padding:.6rem;border:1px solid #d1d5db;border-radius:.5rem}.btn{background:linear-gradient(135deg,#4ade80,#3b82f6);color:white;padding:.7rem;border-radius:.5rem}</style>
    """),
    "utf-8",
)

MISSION_VUE.write_text(
    textwrap.dedent("""\
    <template>
      <div class='p-6 space-y-4'>
        <h2 class='text-xl font-semibold'>Build Your Mission</h2>
        <textarea v-model='mission.mission' placeholder='Mission Statement' class='input'></textarea>
        <textarea v-model='mission.goal' placeholder='Highest Goal' class='input'></textarea>
        <textarea v-model='mission.blocks' placeholder='Emotional Blocks' class='input'></textarea>
        <div class='flex gap-2'>
          <button @click='save' class='btn flex-1'>Save Mission</button>
          <button @click='chat' class='btn flex-1 bg-gray-600'>Chat with Orchestrator</button>
        </div>
      </div>
    </template>
    <script setup>
    import { ref, onMounted } from 'vue'
    const mission = ref({}); onMounted(async()=>{mission.value =
      await fetch('/api/mission').then(r=>r.json())})
    const save = ()=>fetch('/api/mission',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(mission.value)});
    const chat = ()=>window.location='/narrative-chat';  // existing chat route
    </script>
    <style scoped>.input{width:100%;padding:.6rem;border:1px solid #d1d5db;border-radius:.5rem}.btn{background:linear-gradient(135deg,#4ade80,#3b82f6);color:white;padding:.7rem;border-radius:.5rem}</style>
    """),
    "utf-8",
)

ROADMAP_VUE.write_text(
    textwrap.dedent("""\
    <template>
      <div class='p-6'>
        <h2 class='text-xl font-semibold mb-2'>Your Roadmap</h2>
        <p class='text-sm text-gray-500 mb-4'>Let's follow your roadmap to the highest goal.</p>
        <div v-for='m in milestones' :key='m.id'
             :class='[\"roadmap-card\", m.status===\"active\"?\"active\":\"\"]'
             @click='select(m)' >
          <span class='font-medium'>{{ m.title }}</span><br/>
          <small class='text-gray-600'>{{ m.description }}</small>
        </div>
      </div>
    </template>
    <script setup>
    import { ref, onMounted } from 'vue'
    const milestones = ref([])
    onMounted(()=>refresh())
    function refresh(){fetch('/api/roadmap').then(r=>r.json()).then(d=>milestones.value=d)}
    function select(m){if(m.status!=='active')return;
      // maybe open detailed page later
    }
    </script>
    <style scoped>
    .roadmap-card{padding:1rem;border:1px solid #e5e7eb;border-radius:.7rem;margin-bottom:.7rem;cursor:pointer}
    .roadmap-card.active{border-color:#3b82f6;background:#dbeafe}
    </style>
    """),
    "utf-8",
)

# ────────────────────────────────────────────
# 6.  Vue entrypoint (App.vue) simplified
# ────────────────────────────────────────────
app_vue = ROOT / "static/src/App.vue"
app_vue.parent.mkdir(parents=True, exist_ok=True)
app_vue.write_text(
    textwrap.dedent("""\
    <template>
      <BaseHeader/>
      <component :is='current'/>
      <BaseFooter :active='tab' @change='setTab'/>
    </template>
    <script setup>
    import { ref, shallowRef } from 'vue'
    import BaseHeader     from './ui/components/BaseHeader.vue'
    import BaseFooter     from './ui/components/BaseFooter.vue'
    import IdentityView   from './ui/components/IdentityView.vue'
    import MissionView    from './ui/components/MissionView.vue'
    import RoadmapView    from './ui/components/RoadmapView.vue'
    const tab = ref('Identity')
    const comp = {Identity:IdentityView,Mission:MissionView,Roadmap:RoadmapView}
    const current = shallowRef(comp[tab.value])
    function setTab(t){tab.value=t;current.value=comp[t]}
    </script>
    """),
    "utf-8",
)

# ────────────────────────────────────────────
# 7.  Run npm build (if package.json exists)
# ────────────────────────────────────────────
if (ROOT/"package.json").exists():
    subprocess.run(["npm","install"], check=True)
    subprocess.run(["npm","run","build"], check=True)

print("✅   Migration script finished.  Next steps:")
print("   • run:  flask db migrate && flask db upgrade")
print("   • restart the dev server and enjoy the new UX 🚀")
```

**What the script does**

1. **Appends DB & API patches** for profile / mission / roadmap.
2. **Adds a tiny helper** to auto-build the default roadmap.
3. **Creates Vue single-file components** that precisely match your green-blue gradient mock (header), pastel tabs, and scrollable lifeline.
4. Leaves users talking *only* to the Orchestrator; backend silently swaps agents per milestone.

Run the script → run DB migration → `npm run build` → restart Flask.
You should now see the cleaner, mission-focused Missiora you mocked in Figma/DALL·E.

Happy shipping!
